// Flutter (Dart) での判定ロジック例 import 'package:flutter/material.dart'; Color getScheduleColor(Duration remaining) { if (remaining.inMinutes > 60) { return Colors.blue; } else if (remaining.inMinutes > 30) { return Colors.amber; // 黄色 } else if (remaining.inMinutes > 0) { return Colors.red; } else { return Colors.grey; // 終了後 } } // プログレスバー（0.0〜1.0）の割合計算（例: 60分を基準とした場合） double getProgressValue(Duration remaining) { if (remaining.inMinutes <= 0) return 0.0; if (remaining.inMinutes >= 60) return 1.0; return remaining.inMinutes / 60.0; }
